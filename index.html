class OrderController extends Controller
{
  public function dispatch(Request $req)
  {
    $order = Order::create([
      'tenant_id'  => tenant()->id(),
      'driver_id'  => $req->driver_id(),
      'status'     => 'queued',
    ]);

    DispatchOrderJob::dispatch($order)
      ->onQueue('orders')
      ->delay(now()->addSeconds(3));

    return response()->json([
      'status' => 'dispatched',
      'ulid'   => $order->ulid(),
    ]);
  }
}
